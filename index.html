Apps Script code
Paste into Extensions → Apps Script
const SHEET_ID = "YOUR_SHEET_ID_HERE"; const SHEETS = { income:"Income", expense:"Expenses", invest:"Investments" }; const HEADERS = { income: ["Date","Source","Description","Amount","Month","Year","Notes"], expense: ["Date","Category","Description","Amount","Month","Year"], invest: ["Date","Type","Name","Amount","Units","Month","Year","Notes"] }; function doGet(e) { return handle(e); } function doPost(e) { return handle(e); } function handle(e) { const action = (e.parameter||{}).action||""; const ss = SpreadsheetApp.openById(SHEET_ID); Object.keys(SHEETS).forEach(key => { let sh = ss.getSheetByName(SHEETS[key]); if (!sh) { sh = ss.insertSheet(SHEETS[key]); sh.appendRow(HEADERS[key]); sh.setFrozenRows(1); } }); if (action==="ping") return out({ok:true,message:"Connected!"}); if (action==="add") { const d = JSON.parse(e.postData.contents); const dt = new Date(d.date); const m=dt.getMonth()+1; const y=dt.getFullYear(); const sh = ss.getSheetByName(SHEETS[d.type]); if (!sh) return out({ok:false,error:"Unknown type"}); if (d.type==="income") sh.appendRow([d.date,d.source,d.description,Number(d.amount),m,y,d.notes||""]); if (d.type==="expense") sh.appendRow([d.date,d.category,d.description,Number(d.amount),m,y]); if (d.type==="invest") sh.appendRow([d.date,d.itype,d.name,Number(d.amount),d.units||"",m,y,d.notes||""]); return out({ok:true}); } if (action==="fetch") { const result={}; Object.keys(SHEETS).forEach(key => { const sh=ss.getSheetByName(SHEETS[key]); const vals=sh.getDataRange().getValues(); if(vals.length<=1){result[key]=[];return;} const h=vals[0]; result[key]=vals.slice(1).map(r=>{let o={};h.forEach((k,i)=>o[k]=r[i]);return o;}); }); return out({ok:true,data:result}); } return out({ok:false,error:"Unknown action"}); } function out(d){return ContentService.createTextOutput(JSON.stringify(d)).setMimeType(ContentService.MimeType.JSON);}
Setup steps
1
Create a Google Sheet at sheets.google.com. Copy the Sheet ID from the URL — the string between /d/ and /edit.
2
Extensions → Apps Script. Delete existing code, paste the script above. Replace YOUR_SHEET_ID_HERE with your Sheet ID. Save.
3
Deploy → New deployment → Web App. Execute as: Me. Who has access: Anyone. Authorise and copy the Web App URL.
4
Paste URL above and tap Test. Your Sheet will auto-create 3 tabs: Income, Expenses, Investments.
5
Install on phone: In Chrome tap the 3-dot menu → "Add to Home screen". On Safari tap Share → "Add to Home Screen". You'll get a home screen icon like a native app.
6
On laptop: Open the same GitHub Pages URL in Chrome. Bookmark it. Works exactly the same — all data syncs through your Sheet.
